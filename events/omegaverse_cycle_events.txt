# ==========================================
# START CYCLE
# ==========================================
omegaverse_cycle.0010 = {
    type = character_event
    hidden = yes

    immediate = {
        remove_character_flag = omegaverse_first_cycle
        
        # alpha
        if = {
            limit = {
                is_alive = yes
                is_adult = yes
                has_trait_with_flag = omegaverse_alpha_gender
  
            }

            remove_character_flag = omegaverse_cycle_peak
            remove_character_flag = omegaverse_cycle_ebbed

            remove_character_flag = omegaverse_in_heat

            add_character_flag = omegaverse_in_rut
            add_character_flag = omegaverse_cycle_start

            add_trait = omegaverse_in_rut_trait

            # first cycle
            if = {
                limit = {
                    NOT = {
                        has_character_flag = omegaverse_had_first_cycle
                    }
                }

                add_character_flag = omegaverse_first_cycle

                trigger_event = {
                    id = omegaverse_cycle.1010
                }
            }
            else = {
                trigger_event = {
                    id = omegaverse_cycle.1012
                }
            }

            trigger_event = {
                    id = omegaverse_cycle.0020
                    months = { 6 8 }
            }
        }

        # omega
        else_if = {
            limit = {
                is_alive = yes
                is_adult = yes

                has_trait_with_flag = omegaverse_omega_gender
            }

            # no cycle for preggers
            if = {
                limit = {
                    OR = {
                        is_pregnant = yes
                        has_trait = omegaverse_carrying_child
                        has_character_flag = omegaverse_pregnancy_pending
                    }
                }

                # reschedule after pregnancy 
                trigger_event = {
                    id = omegaverse_cycle.0010
                    months = { 10 14 }
                }
            }

            else = {
                remove_character_flag = omegaverse_cycle_peak
                remove_character_flag = omegaverse_cycle_ebbed

                remove_character_flag = omegaverse_in_rut

                add_character_flag = omegaverse_in_heat
                add_character_flag = omegaverse_cycle_start

                add_trait = omegaverse_in_heat_trait
                
                # first cycle
                if = {
                    limit = {
                        NOT = {
                            has_character_flag = omegaverse_had_first_cycle
                        }
                    }

                    add_character_flag = omegaverse_first_cycle

                    trigger_event = {
                        id = omegaverse_cycle.1011
                    }
                }
                else = {
                    trigger_event = {
                        id = omegaverse_cycle.1013
                    }
                }

                trigger_event = {
                    id = omegaverse_cycle.0020
                    months = { 6 8 }
                }
            }
        }

        # clean up for changed trait (..eventually)
        else = {
            remove_character_flag = omegaverse_cycle_start
            remove_character_flag = omegaverse_cycle_peak
            remove_character_flag = omegaverse_cycle_ebbed
            
            remove_character_flag = omegaverse_in_heat
            remove_character_flag = omegaverse_in_rut
            remove_character_flag = omegaverse_cycle_chain_active
            
            remove_character_flag = omegaverse_first_cycle
            remove_character_flag = omegaverse_had_first_cycle
        }
    }
}

# ==========================================
# PEAK HEAT / RUT
# ==========================================
omegaverse_cycle.0020 = {
    type = character_event
    hidden = yes

    immediate = {
        if = {
            limit = {
                is_alive = yes
                OR = {
                    has_character_flag = omegaverse_in_heat
                    has_character_flag = omegaverse_in_rut
                }
            }

            remove_character_flag = omegaverse_cycle_start
            add_character_flag = omegaverse_cycle_peak

            # first cycle peak
            if = {
                limit = {
                    has_character_flag = omegaverse_first_cycle
                }

                if = {
                    limit = {
                        has_character_flag = omegaverse_in_rut
                    }

                    trigger_event = {
                        id = omegaverse_cycle.1020
                    }
                }
                else_if = {
                    limit = {
                        has_character_flag = omegaverse_in_heat
                    }

                    trigger_event = {
                        id = omegaverse_cycle.1021
                    }
                }
            }

            trigger_event = {
                id = omegaverse_cycle.0030
                months = { 6 8 }
            }
        }
    }
}

            }
        }
    }
}

# ==========================================
# END CYCLE
# ==========================================
omegaverse_cycle.0040 = {
    type = character_event
    hidden = yes

    immediate = {
        # first cycle completed
        if = {
            limit = {
                has_character_flag = omegaverse_first_cycle
            }

            if = {
                limit = {
                    has_character_flag = omegaverse_in_rut
                }

                trigger_event = {
                    id = omegaverse_cycle.1040
                }
            }
            else_if = {
                limit = {
                    has_character_flag = omegaverse_in_heat
                }

                trigger_event = {
                    id = omegaverse_cycle.1041
                }
            }
    
            remove_character_flag = omegaverse_first_cycle
            add_character_flag = omegaverse_had_first_cycle
        }
        
        remove_character_flag = omegaverse_in_heat
        remove_character_flag = omegaverse_in_rut

        remove_character_flag = omegaverse_cycle_start
        remove_character_flag = omegaverse_cycle_peak
        remove_character_flag = omegaverse_cycle_ebbed

        remove_trait = omegaverse_in_heat_trait    
        remove_trait = omegaverse_in_rut_trait

        if = {
            limit = {
                is_alive = yes
                is_adult = yes

                OR = {
                    has_trait_with_flag = omegaverse_alpha_gender
                    has_trait_with_flag = omegaverse_omega_gender
                }
            }

            # schedule next
            trigger_event = {
                id = omegaverse_cycle.0010
                years = { 3 5 }
            }
        }

        else = {
            remove_character_flag = omegaverse_cycle_chain_active
        }
    }
}
