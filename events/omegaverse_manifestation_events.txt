# 0001 - manifestation
# 0010 - hint processor
# 1001 - child

# ==========================================
# MANIFESTATION PROCESSOR
# ==========================================
omegaverse_manifestation.0001 = {
    type = character_event
    hidden = yes

    trigger = {
        has_trait = omegaverse_unrevealed_secondary_gender
        age >= 16
    }

    immediate = {
        save_scope_as = omegaverse_child
        save_scope_as = child

        omegaverse_reveal_secondary_gender = yes

        if = {
            limit = {
                is_ai = no
            }

            trigger_event = {
                id = omegaverse_manifestation.0002
            }
        }

        else = {
            limit = {
                is_ai = yes
            }

            every_in_list = {
                variable = omegaverse_parents

                limit = {
                    is_ai = no
                }

                trigger_event = {
                    id = omegaverse_manifestation.0003
                }
            }
        }
    }
}

# ==========================================
# HINT PROCESSOR
# ==========================================
omegaverse_manifestation.4000 = {
    type = character_event
    hidden = yes

    trigger = {
        has_trait = omegaverse_unrevealed_secondary_gender

        NOT = {
            has_character_flag = omegaverse_hint_received
        }
    }

    immediate = {
        save_scope_as = omegaverse_child
        save_scope_as = child

        omegaverse_ensure_valid_destiny = yes
        omegaverse_give_manifestation_hint = yes
        
        # child player
        if = {
            limit = {
                is_ai = no
            }

            random_list = {
                50 = {
                    trigger_event = {
                        id = omegaverse_manifestation.4010
                    }
                }
                50 = {
                    trigger_event = {
                        id = omegaverse_manifestation.4011
                    }
                }
            }
        }

        else = {
            every_in_list = {
                variable = omegaverse_parents

                limit = {
                    is_ai = no
                }

                random_list = {
                    50 = {
                        trigger_event = {
                            id = omegaverse_manifestation.4020
                        }
                    }
                    50 = {
                        trigger_event = {
                            id = omegaverse_manifestation.4021
                        }
                    }
                }
            }
        }
    }
}
