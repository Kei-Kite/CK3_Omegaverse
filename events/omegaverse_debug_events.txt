        clear_saved_scope = debug_sex_partner
        clear_saved_scope = debug_test_bearer
        clear_saved_scope = debug_test_sire

        scope:omegaverse_debug_target = {
            ordered_memory = {
                limit = {
                    has_memory_type = had_sex

                    any_memory_participant = {
                        this != scope:omegaverse_debug_target
                    }
                }

                order_by = memory_creation_date
                position = 0

                save_scope_as = debug_unchecked_sex_memory
            }
        }

        remove_variable = debug_sex_memory_age_days

        if = {
            limit = {
                exists = scope:debug_unchecked_sex_memory
            }

            set_variable = {
                name = debug_sex_memory_age_days
                value = {
                    value = current_date
                    subtract = scope:debug_unchecked_sex_memory.memory_creation_date
                }
            }

            scope:debug_unchecked_sex_memory = {
                random_memory_participant = {
                    limit = {
                        this != scope:omegaverse_debug_target
                    }

                    save_scope_as = debug_sex_partner
                }
            }
        }

        clear_saved_scope = debug_pregnancy_partner

        if = {
            limit = {
                exists = scope:omegaverse_debug_target.var:omegaverse_sire
            }

            scope:omegaverse_debug_target.var:omegaverse_sire = {
                save_scope_as = debug_pregnancy_partner
            }

            limit = {
                exists = scope:debug_sex_partner

                    omegaverse_can_sire_trigger = yes

                save_scope_as = debug_test_sire

                exists = scope:debug_sex_partner

                    omegaverse_can_bear_trigger = yes

                    omegaverse_can_sire_trigger = yes
