# ==========================================
# EVENT INDEX
#
# 1000 - omegaverse newborn
# 1001 - naming window
# 2001 - birth management
# 3001 - child dies during delivery
# 3002 - sire: child dies during delivery
# 3011 - bearer dies during delivery
# 3012 - sire: bearer dies during delivery
# 3021 - bearer and child die
# 3022 - sire: bearer and child die
# 3031 - child becomes sickly
# 3032 - sire: child becomes sickly
# 3033 - sickly child recovery management
# 3034 - sickly child recovers
# 3035 - bearer/sire: sickly child recovers
# 3041 - bearer becomes ill
# ==========================================

# ==========================================
# SCOPES
#   
# scope:omegaverse_bearer 
# scope:omegaverse_sire 
# scope:child   (vanilla)
# scope:dummy_gender 
# var:omegaverse_child 
#   bearer - pointer to latest child
# ==========================================

# ==========================================
# OMEGAVERSE BIRTH
#
# creates child
# vanilla parentage gives bearer the
# fitting relationship.
# omegaverse parents are separate
# ==========================================
omegaverse_birth.1000 = {
    type = character_event
    hidden = yes

    trigger = {
        has_trait = omegaverse_carrying_child
    }

    immediate = {
        remove_trait = omegaverse_carrying_child

        save_scope_as = omegaverse_bearer

        if = {
            limit = {
                exists = var:omegaverse_sire
            }

            var:omegaverse_sire = {
                save_scope_as = omegaverse_sire
            }
        }

        # gender taken from conception
        if = {
            limit = {
                has_character_flag = omegaverse_unborn_female
            }

            create_character = {
                age = 0
                gender = female
                faith = root.faith
                culture = root.culture
                location = root.location

                save_scope_as = child
            }
        }

        else = {
            create_character = {
                age = 0
                gender = male
                faith = root.faith
                culture = root.culture
                location = root.location

                save_scope_as = child
            }
        }

        # parentage
        scope:child = {

            add_character_flag = omegaverse_child

            set_variable = {
                name = omegaverse_bearer
                value = scope:omegaverse_bearer
            }

            if = {
                limit = {
                    exists = scope:omegaverse_sire
                }

                set_variable = {
                    name = omegaverse_sire
                    value = scope:omegaverse_sire
                }
            }

            # assign child to bearer
            # biggest wip bc REAL father n stuff
            if = {
                limit = {
                    scope:omegaverse_bearer = {
                        is_female = yes
                    }
                }

                set_mother = scope:omegaverse_bearer
            }

            else = {
                set_father = scope:omegaverse_bearer
            }

            clear_variable_list = omegaverse_parents

            # for gui
            add_to_variable_list = {
                name = omegaverse_parents
                target = scope:omegaverse_bearer
            }

            if = {
                limit = {
                    exists = scope:omegaverse_sire
                }

                add_to_variable_list = {
                    name = omegaverse_parents
                    target = scope:omegaverse_sire
                }
            }
        }

        set_variable = {
            name = omegaverse_child
            value = scope:child
        }

        # give parents access to child
        scope:omegaverse_bearer = {
            add_to_variable_list = {
                name = omegaverse_children
                target = scope:child
            }
        }

        if = {
            limit = {
                exists = scope:omegaverse_sire
            }

            scope:omegaverse_sire = {
                add_to_variable_list = {
                    name = omegaverse_children
                    target = scope:child
                }
            }
        }

        # house
        scope:child = {

            if = {
                limit = {
                    exists = scope:omegaverse_sire
                    exists = scope:omegaverse_sire.house
                }

                set_house = scope:omegaverse_sire.house
            }

            else_if = {
                limit = {
                    exists = scope:omegaverse_bearer.house
                }

                set_house = scope:omegaverse_bearer.house
            }
        }

        # employer
        scope:child = {
            set_employer = scope:omegaverse_bearer
        }

        # genetics
        scope:child = {

            omegaverse_determine_abo = yes

            add_trait = omegaverse_unrevealed_secondary_gender
            add_character_flag = omegaverse_secondary_gender_hidden
        }

        # child ill
        if = {
            limit = {
                has_character_flag = omegaverse_child_will_become_sickly
            }

            remove_character_flag = omegaverse_child_will_become_sickly

            trigger_event = {
                id = omegaverse_birth.3031
                days = { 10 30 }
            }
        }

        # bearer ill
        if = {
            limit = {
                has_character_flag = omegaverse_bearer_will_become_ill
            }

            remove_character_flag = omegaverse_bearer_will_become_ill

            trigger_event = {
                id = omegaverse_birth.3041
                days = { 10 30 }
            }
        }

        # bearer dies
        if = {
            limit = {
                has_character_flag = omegaverse_bearer_will_die
            }

            trigger_event = {
                id = omegaverse_birth.3011
            }
        }

        # birth event
        else_if = {
            limit = {
                is_ai = no
            }

            trigger_event = {
                id = omegaverse_birth.1001
            }
        }

        omegaverse_clear_pregnancy = yes
    }
}

# ==========================================
# BIRTH MANAGEMENT
#
# only gives birth if the pregnancy is old
# enough 
# ==========================================
omegaverse_birth.2001 = {
    type = character_event
    hidden = yes

    trigger = {
        has_trait = omegaverse_carrying_child

        OR = {
            NOT = {
                exists = var:omegaverse_pregnancy_start
            }

            current_date >= {
                value = var:omegaverse_pregnancy_start
                add = 200
            }
        }
    }

    immediate = {
        trigger_event = {
            id = omegaverse_birth.1000
        }
    }
}

omegaverse_birth.3001 = {
    type = character_event
    title = birth.3001.t

    desc = birth.3001.desc_other_child
    theme = death

    override_background = {
        reference = bedchamber
    }

    left_portrait = {
        character = root
        animation = grief
        outfit_tags = { nightgown }
    }

    trigger = {
        has_trait = omegaverse_carrying_child
    }

    immediate = {
        if = {
            limit = {
                has_character_flag = omegaverse_unborn_female
            }

            dummy_female = {
                save_scope_as = dummy_gender
            }
        }

        else = {
            dummy_male = {
                save_scope_as = dummy_gender
            }
        }

        save_scope_as = omegaverse_bearer
        save_scope_as = mother

        play_music_cue = mx_cue_death

        create_character_memory = {
            type = child_stillborn
            participants = {
                mother = root
            }
        }

        if = {
            limit = {
                exists = var:omegaverse_sire

                var:omegaverse_sire = {
                    is_alive = yes
                }
            }

            var:omegaverse_sire = {
                trigger_event = {
                    id = omegaverse_birth.3002
                }
            }
        }

        omegaverse_clear_pregnancy = yes

        custom_tooltip = birth.3001.a.tt
    }

    option = {
        name = birth.3001.a
    }
}

    type = character_event
    title = birth.3001.t

    }

        has_character_flag = omegaverse_bearer_will_die

                    }
                }
            }

            var:omegaverse_child = {
                var:omegaverse_sire = {

                        id = omegaverse_birth.3012
                    }
                }
            }
        }

        # the one flag the clean-up leaves alone
        remove_character_flag = omegaverse_bearer_will_die

        omegaverse_clear_pregnancy = yes
    }

    option = {
        name = birth.3011.a

        death = {
            death_reason = death_childbirth
        }
    }
