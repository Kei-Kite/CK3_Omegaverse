omegaverse_inherit_congenital_traits_effect = {

# ==========================================
# INTELLECT
# ==========================================
omegaverse_inherit_intellect_effect = {
    if = {
        limit = {
            OR = {
                scope:omegaverse_bearer = { has_trait = intellect_good }
                scope:omegaverse_sire = { has_trait = intellect_good }
            }
        }

        # tier 3
        set_variable = {
            name = omegaverse_inheritance_chance
            value = 0
        }

        if = {
            limit = {
                scope:omegaverse_bearer = { has_trait = intellect_good_3 }
                scope:omegaverse_sire = { has_trait = intellect_good_3 }
            }
            set_variable = { name = omegaverse_inheritance_chance value = 80 }
        }
        else_if = {
            limit = {
                OR = {
                    scope:omegaverse_bearer = { has_trait = intellect_good_3 }
                    scope:omegaverse_sire = { has_trait = intellect_good_3 }
                }
            }
            set_variable = { name = omegaverse_inheritance_chance value = 40 }
        }
        else_if = {
            limit = {
                OR = {
                    scope:omegaverse_bearer = { has_trait = intellect_good_2 }
                    scope:omegaverse_sire = { has_trait = intellect_good_2 }
                }
            }
            set_variable = { name = omegaverse_inheritance_chance value = 15 }
        }
        else_if = {
            limit = {
                OR = {
                    scope:omegaverse_bearer = { has_trait = intellect_good_1 }
                    scope:omegaverse_sire = { has_trait = intellect_good_1 }
                }
            }
            set_variable = { name = omegaverse_inheritance_chance value = 5 }
        }

        random = {
            chance = var:omegaverse_inheritance_chance
            add_trait = intellect_good_3
        }

        # tier 2
        if = {
            limit = { NOT = { has_trait = intellect_good_3 } }

            set_variable = {
                name = omegaverse_inheritance_chance
                value = 0
            }

            if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_3 }
                        scope:omegaverse_sire = { has_trait = intellect_good_3 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 30 }
            }
            else_if = {
                limit = {
                    AND = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_2 }
                        scope:omegaverse_sire = { has_trait = intellect_good_2 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 60 }
            }
            else_if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_2 }
                        scope:omegaverse_sire = { has_trait = intellect_good_2 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 30 }
            }
            else_if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_1 }
                        scope:omegaverse_sire = { has_trait = intellect_good_1 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 10 }
            }

            random = {
                chance = var:omegaverse_inheritance_chance
                add_trait = intellect_good_2
            }
        }

        # tier 1
        if = {
            limit = {
                NOT = { has_trait = intellect_good_3 }
                NOT = { has_trait = intellect_good_2 }
            }

            set_variable = {
                name = omegaverse_inheritance_chance
                value = 0
            }

            if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_3 }
                        scope:omegaverse_sire = { has_trait = intellect_good_3 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 15 }
            }
            else_if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_2 }
                        scope:omegaverse_sire = { has_trait = intellect_good_2 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 30 }
            }
            else_if = {
                limit = {
                    OR = {
                        scope:omegaverse_bearer = { has_trait = intellect_good_1 }
                        scope:omegaverse_sire = { has_trait = intellect_good_1 }
                    }
                }
                set_variable = { name = omegaverse_inheritance_chance value = 60 }
            }

            random = {
                chance = var:omegaverse_inheritance_chance
                add_trait = intellect_good_1
            }
        }

        remove_variable = omegaverse_inheritance_chance
    }

    # bad branch
    else_if = {
        limit = {
            OR = {
                scope:omegaverse_bearer = { has_trait = intellect_bad }
                scope:omegaverse_sire = { has_trait = intellect_bad }
            }
        }

        random = {
            chance = 20
            add_trait = intellect_bad_3
        }
        else = {
            random = {
                chance = 25
                add_trait = intellect_bad_2
            }
            else = {
                random = {
                    chance = 33
                    add_trait = intellect_bad_1
                }
            }
        }
    }
}
