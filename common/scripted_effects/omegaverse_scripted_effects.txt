# ======================================
# TRAIT GENETICS
#
# Pair:  A / B / O  % for child
# A+O   40% / 20% / 40%
# A+B   55% / 35% / 10%
# B+O   10% / 35% / 55%
# B+B   10% / 80% / 10%
# ======================================
omegaverse_determine_abo = {
    omegaverse_clear_destiny = yes

    add_character_flag = omegaverse_secondary_gender_hidden

    # parent gender checks
    if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_alpha_gender
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_omega_gender
            }
        }

        random_list = {
            40 = {
                add_character_flag = omegaverse_destined_alpha
            }
            20 = {
                add_character_flag = omegaverse_destined_beta
            }
            40 = {
                add_character_flag = omegaverse_destined_omega
            }
        }
    }

    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_alpha_gender
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_beta_gender
            }
        }

        random_list = {
            55 = {
                add_character_flag = omegaverse_destined_alpha
            }
            35 = {
                add_character_flag = omegaverse_destined_beta
            }
            10 = {
                add_character_flag = omegaverse_destined_omega
            }
        }
    }

    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_beta_gender
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_omega_gender
            }
        }

        random_list = {
            10 = {
                add_character_flag = omegaverse_destined_alpha
            }
            35 = {
                add_character_flag = omegaverse_destined_beta
            }
            55 = {
                add_character_flag = omegaverse_destined_omega
            }
        }
    }

    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_beta_gender
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_beta_gender
            }
        }

        random_list = {
            10 = {
                add_character_flag = omegaverse_destined_alpha
            }
            80 = {
                add_character_flag = omegaverse_destined_beta
            }
            10 = {
                add_character_flag = omegaverse_destined_omega
            }
        }
    }

    # fallback
    else = {
        omegaverse_basic_abo_roll = yes
    }

    # subtype
    omegaverse_determine_subtype = yes
}

# ======================================
# SUBTYPE GENETICS
#
# Pair:  D / S / R  % for child
# D+D   80% / 15% / 5%
# D+R   25% / 50% / 25%
# R+R   5%  / 15% / 80%
# D+S   45% / 50% / 5%
# R+S   5%  / 50% / 45%
# S+S   10% / 80% / 10%
# ======================================
omegaverse_determine_subtype = {
    # dom + dom
    if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_dominant_subtrait
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_dominant_subtrait
            }
        }

        random_list = {
            80 = {
                add_character_flag = omegaverse_destined_dominant
            }
            15 = {
                add_character_flag = omegaverse_destined_standard
            }
            5 = {
                add_character_flag = omegaverse_destined_recessive
            }
        }
    }

    # dom + rec
    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            OR = {
                AND = {
                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_dominant_subtrait
                    }

                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_recessive_subtrait
                    }
                }

                AND = {
                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_recessive_subtrait
                    }

                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_dominant_subtrait
                    }
                }
            }
        }

        random_list = {
            25 = {
                add_character_flag = omegaverse_destined_dominant
            }
            50 = {
                add_character_flag = omegaverse_destined_standard
            }
            25 = {
                add_character_flag = omegaverse_destined_recessive
            }
        }
    }

    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            var:omegaverse_sire = {
                has_trait_with_flag = omegaverse_recessive_subtrait
            }

            var:omegaverse_bearer = {
                has_trait_with_flag = omegaverse_recessive_subtrait
            }
        }

        random_list = {
            5 = {
                add_character_flag = omegaverse_destined_dominant
            }
            15 = {
                add_character_flag = omegaverse_destined_standard
            }
            80 = {
                add_character_flag = omegaverse_destined_recessive
            }
        }
    }

    # dom + stan
    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            OR = {
                AND = {
                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_dominant_subtrait
                    }

                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_standard_subtrait
                    }
                }

                AND = {
                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_dominant_subtrait
                    }

                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_standard_subtrait
                    }
                }
            }
        }

        random_list = {
            45 = {
                add_character_flag = omegaverse_destined_dominant
            }
            50 = {
                add_character_flag = omegaverse_destined_standard
            }
            5 = {
                add_character_flag = omegaverse_destined_recessive
            }
        }
    }

    # rec + stan
    else_if = {
        limit = {
            exists = var:omegaverse_sire
            exists = var:omegaverse_bearer

            OR = {
                AND = {
                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_recessive_subtrait
                    }

                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_standard_subtrait
                    }
                }

                AND = {
                    var:omegaverse_bearer = {
                        has_trait_with_flag = omegaverse_recessive_subtrait
                    }

                    var:omegaverse_sire = {
                        has_trait_with_flag = omegaverse_standard_subtrait
                    }
                }
            }
        }

        random_list = {
            5 = {
                add_character_flag = omegaverse_destined_dominant
            }
            50 = {
                add_character_flag = omegaverse_destined_standard
            }
            45 = {
                add_character_flag = omegaverse_destined_recessive
            }
        }

# ======================================
# ENSURE ABO ASSIGNMENT
# 
# fallback
# only if char doesnt have the hidden 
# trait OR any gender trait (flag)
# ======================================
omegaverse_ensure_abo = {
    if = {
        limit = {
            age >= 16
            NOT = {
                has_trait_with_flag = omegaverse_secondary_gender
            }
        }

        omegaverse_reveal_secondary_gender = yes
    }
    else_if = {
        limit = {
            age < 16

            NOT = {
                has_trait_with_flag = omegaverse_secondary_gender
            }

            NOT = {
                has_trait = omegaverse_unrevealed_secondary_gender
            }
        }

        omegaverse_determine_abo = yes
        add_trait = omegaverse_unrevealed_secondary_gender
        add_character_flag = omegaverse_secondary_gender_hidden
    }

    if = {
        limit = {
            age >= 16
            AND = {
                has_trait_with_flag = omegaverse_secondary_gender
                has_trait = omegaverse_unrevealed_secondary_gender
            }
        }
        
        remove_trait = omegaverse_unrevealed_secondary_gender
    }
}

# ======================================
# DESTINY APPLICATION
# 
# does not remove anything!
# ======================================
omegaverse_apply_destined_abo = {
    if = {
        limit = {
            has_character_flag = omegaverse_destined_alpha
            has_character_flag = omegaverse_destined_dominant
        }

        add_trait = omegaverse_dominant_alpha
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_alpha
            has_character_flag = omegaverse_destined_recessive
        }

        add_trait = omegaverse_recessive_alpha
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_alpha
            has_character_flag = omegaverse_destined_standard
            
        }

        add_trait = omegaverse_standard_alpha
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_beta
            OR = {
                has_character_flag = omegaverse_destined_dominant
                has_character_flag = omegaverse_destined_recessive
            }
        }

        add_trait = omegaverse_unstable_beta
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_beta
            has_character_flag = omegaverse_destined_standard
        }

        add_trait = omegaverse_standard_beta
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_omega
            has_character_flag = omegaverse_destined_dominant
        }

        add_trait = omegaverse_dominant_omega
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_omega
            has_character_flag = omegaverse_destined_recessive
        }

        add_trait = omegaverse_recessive_omega
    }

    else_if = {
        limit = {
            has_character_flag = omegaverse_destined_omega
            has_character_flag = omegaverse_destined_standard
        }

        add_trait = omegaverse_standard_omega
    }
}

# ======================================
# VALIDATE DESTINY
# ======================================
omegaverse_ensure_valid_destiny = {
    if = {
        limit = {
            NOT = {
                OR = {
                    has_character_flag = omegaverse_destined_alpha
                    has_character_flag = omegaverse_destined_beta
                    has_character_flag = omegaverse_destined_omega
                }
            }
        }

        omegaverse_determine_abo = yes
    }

    else_if = {
        limit = {
            NOT = {
                OR = {
                    has_character_flag = omegaverse_destined_dominant
                    has_character_flag = omegaverse_destined_standard
                    has_character_flag = omegaverse_destined_recessive
                }
            }
        }

        omegaverse_determine_subtype = yes
    }
}

# ======================================
# GENDER REVEAL
# ======================================
omegaverse_reveal_secondary_gender = {
    omegaverse_ensure_valid_destiny = yes
    
    omegaverse_apply_destined_abo = yes 
    
    remove_trait = omegaverse_unrevealed_secondary_gender

    omegaverse_clear_destiny = yes

    add_character_flag = omegaverse_secondary_gender_revealed

    omegaverse_update_pregnancy_suppression = yes

    omegaverse_schedule_cycle = yes
}

# ======================================
# ASSIGN MANIFESTATION HINT
# ======================================
omegaverse_give_manifestation_hint = {
    if = {
        limit = {
            NOT = {
                has_character_flag = omegaverse_hint_received
            }
        }

        add_character_flag = omegaverse_hint_received

        if = {
            limit = {
                has_character_flag = omegaverse_destined_alpha
                has_character_flag = omegaverse_destined_dominant
            }

            add_character_flag = omegaverse_hint_dominant_alpha
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_alpha
                has_character_flag = omegaverse_destined_recessive
            }

            add_character_flag = omegaverse_hint_recessive_alpha
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_alpha
                has_character_flag = omegaverse_destined_standard
            }

            add_character_flag = omegaverse_hint_standard_alpha
        }

        # unstable beta
        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_beta
                OR = {
                    has_character_flag = omegaverse_destined_dominant
                    has_character_flag = omegaverse_destined_recessive
                }
            }

            add_character_flag = omegaverse_hint_unstable_beta
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_beta
                has_character_flag = omegaverse_destined_standard
            }

            add_character_flag = omegaverse_hint_standard_beta
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_omega
                has_character_flag = omegaverse_destined_dominant
            }

            add_character_flag = omegaverse_hint_dominant_omega
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_omega
                has_character_flag = omegaverse_destined_recessive
            }

            add_character_flag = omegaverse_hint_recessive_omega
        }

        else_if = {
            limit = {
                has_character_flag = omegaverse_destined_omega
                has_character_flag = omegaverse_destined_standard
            }

            add_character_flag = omegaverse_hint_standard_omega
        }
    }
}

# ==========================================
# CYCLE SCHEDULER
# ==========================================
omegaverse_schedule_cycle = {
    if = {
        limit = {
            is_alive = yes
            is_adult = yes

            NOT = {
                has_character_flag = omegaverse_cycle_chain_active
            }

            OR = {
                has_trait = omegaverse_dominant_alpha
                has_trait = omegaverse_standard_alpha
                has_trait = omegaverse_recessive_alpha

                has_trait = omegaverse_dominant_omega
                has_trait = omegaverse_standard_omega
                has_trait = omegaverse_recessive_omega
            }
        }

        add_character_flag = omegaverse_cycle_chain_active

        trigger_event = {
            id = omegaverse_cycle.0010
            months = { 6 18 }
        }
    }
}

# ======================================
# REMOVE DESTINY FLAGS
# ======================================
omegaverse_clear_destiny = {
    remove_character_flag = omegaverse_destined_alpha
    remove_character_flag = omegaverse_destined_beta
    remove_character_flag = omegaverse_destined_omega

    remove_character_flag = omegaverse_destined_dominant
    remove_character_flag = omegaverse_destined_recessive
    remove_character_flag = omegaverse_destined_standard

    remove_character_flag = omegaverse_hint_received
    remove_character_flag = omegaverse_hint_dominant_alpha
    remove_character_flag = omegaverse_hint_standard_alpha
    remove_character_flag = omegaverse_hint_recessive_alpha
    remove_character_flag = omegaverse_hint_unstable_beta
    remove_character_flag = omegaverse_hint_standard_beta
    remove_character_flag = omegaverse_hint_dominant_omega
    remove_character_flag = omegaverse_hint_standard_omega
    remove_character_flag = omegaverse_hint_recessive_omega

    remove_character_flag = omegaverse_secondary_gender_hidden
}
